def balanced(ss,n):
    if n==0:
        return []
    if len(ss)<n:
        return []

    cx=sum(s[1] for s in ss)/len(ss)
    cy=sum(s[2] for s in ss)/len(ss)
    rem=ss[:]

    a=min(
        rem,
        key=lambda s:(
            abs(s[1]-cx)+abs(s[2]-cy),
            s[0],
            s[1],
            s[2]
        )
    )
    out=[a]
    rem.remove(a)

    def dist(x,y):
        return (
            abs(x[0]-y[0])*4+
            abs(x[1]-y[1])+
            abs(x[2]-y[2])
        )

    while rem and len(out)<n:
        z=max(
            rem,
            key=lambda s:min(dist(s,a) for a in out)
        )
        out.append(z)
        rem.remove(z)

    return out

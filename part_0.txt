def parse_rows(text):

    r'(?:component=|package=)([^/;,\s]+)',  
    intent  
)  

if m:  
    return m.group(1)  

m = re.search(  
    r'package:([^/;,\s]+)',  
    intent  
)  

if m:  
    return m.group(1)  

return None

def parse_usage(text):
result = {}

pieces = re.split(  
    r'(?=package=[^\s]+)',  
    text  
)  

for piece in pieces:  
    m = re.search(  
        r'package=([^\s]+)',  
        piece  
    )  

    if not m:  
        continue  

    package = m.group(1).rstrip(",")  

    if "." not in package:  
        continue  

    vals = {}  

    for key in (  
        "totalTimeInForeground",  
        "totalTimeForeground",  
        "totalTimeUsed",  
        "lastTimeUsed",  
        "lastTimeForeground",  
        "appLaunchCount"  
    ):  
        mm = re.search(  
            rf'{key}=([0-9]+)',  
            piece  
        )  

        if mm:  
            vals[key] = int(mm.group(1))  

    foreground = max(  
        vals.get("totalTimeInForeground", 0),  
        vals.get("totalTimeForeground", 0),  
        vals.get("totalTimeUsed", 0)  
    )  

    last_used = max(  
        vals.get("lastTimeUsed", 0),  
        vals.get("lastTimeForeground", 0)  
    )  

    launches = vals.get(  
        "appLaunchCount",  
        0  
    )  

    old = result.get(  
        package,  
        {  
            "time_ms": 0,  
            "last_ms": 0,  
            "launches": 0  
        }  
    )  

    old["time_ms"] = max(  
        old["time_ms"],  
        foreground  
    )  

    old["last_ms"] = max(  
        old["last_ms"],  
        last_used  
    )  

    old["launches"] = max(  
        old["launches"],  
        launches  
    )  

    result[package] = old  

return result

def classify(label, package):

f"{label or ''} {package or ''}"

def score_usage(usage, now_ms):
if not usage:
return 0.0

time_minutes = max(  
    0,  
    usage.get("time_ms", 0)  
) / 60000.0  

time_score = min(  
    1.0,  
    math.log1p(time_minutes) /  
    math.log1p(600)  
)  

launches = max(  
    0,  
    usage.get("launches", 0)  
)  

launch_score = min(  
    1.0,  
    math.log1p(launches) /  
    math.log1p(100)  
)  

last = usage.get(  
    "last_ms",  
    0  
)  

if last > 0 and last <= now_ms:  
    age_days = (  
        now_ms - last  
    ) / 86400000.0  

    recency_score = math.exp(  
        -max(0, age_days) / 14.0  
    )  
else:  
    recency_score = 0.0  

w = DEFAULTS["weights"]  

return round(  
    w["time"] * time_score  
    + w["launches"] * launch_score  
    + w["recency"] * recency_score,  
    4  
)

def parse_ui_nodes(xml):
nodes = []

for match in re.finditer(  
    r'<node\b([^>]*)/>',  
    xml  
):  
    attrs = dict(  
        re.findall(  
            r'(\w+)="([^"]*)"',  
            match.group(1)  
        )  
    )  

    text = attrs.get("text", "")  
    desc = attrs.get(  
        "content-desc",  
        ""  
    )  
    bounds = attrs.get(  
        "bounds",  
        ""  
    )  

    bm = re.search(  
        r'(\d+),(\d+)(\d+),(\d+)',  
        bounds  
    )  

    if not bm:  
        continue  

    x1, y1, x2, y2 = map(  
        int,  
        bm.groups()  
    )  

    if not text and not desc:  
        continue  

    nodes.append({  
        "text": text,  
        "desc": desc,  
        "bounds": bounds,  
        "center": [  
            (x1 + x2) // 2,  
            (y1 + y2) // 2  
        ]  
    })  

return nodes

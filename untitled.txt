    r'(?:component=|package=)([^/;,\s]+)',  
    intent  
)

    return m.group(1)  

m = re.search(  
    r'package:([^/;,\s]+)',  
    intent  
)

return None

def parse_usage(text):
result = {}

pieces = re.split(  
    r'(?=package=[^\s]+)',  
    text  
)  

for piece in pieces:  
    m = re.search(  
        r'package=([^\s]+)',  
        piece  
    )  

    if not m:  
        continue  

    package = m.group(1).rstrip(",")  

    if "." not in package:  
        continue  

    vals = {}  

    for key in (  
        "totalTimeInForeground",  
        "totalTimeForeground",  
        "totalTimeUsed",  
        "lastTimeUsed",  
        "lastTimeForeground",  
        "appLaunchCount"  
    ):  
        mm = re.search(  
            rf'{key}=([0-9]+)',  
            piece  
        )  

        if mm:  
            vals[key] = int(mm.group(1))  

    foreground = max(  
        vals.get("totalTimeInForeground", 0),  
        vals.get("totalTimeForeground", 0),  
        vals.get("totalTimeUsed", 0)  
    )  

    last_used = max(  
        vals.get("lastTimeUsed", 0),  
        vals.get("lastTimeForeground", 0)  
    )  

    launches = vals.get(  
        "appLaunchCount",  
        0  
    )  

    old = result.get(  
        package,  
        {  
            "time_ms": 0,  
            "last_ms": 0,  
            "launches": 0  
        }  
    )  

    old["time_ms"] = max(  
        old["time_ms"],  
        foreground  
    )  

    old["last_ms"] = max(  
        old["last_ms"],  
        last_used  
    )  

    old["launches"] = max(  
        old["launches"],  
        launches  
    )  

    result[package] = old  

return result

def classify(label, package):

f"{label or ''} {package or ''}"
).lower()

def score_usage(usage, now_ms):
if not usage:
return 0.0

time_minutes = max(  
    0,  
    usage.get("time_ms", 0)  
) / 60000.0  

time_score = min(  
    1.0,  
    math.log1p(time_minutes) /  
    math.log1p(600)  
)  

launches = max(  
    0,  
    usage.get("launches", 0)  
)  

launch_score = min(  
    1.0,  
    math.log1p(launches) /  
    math.log1p(100)  
)  

last = usage.get(  
    "last_ms",  
    0  
)  

if last > 0 and last <= now_ms:  
    age_days = (  
        now_ms - last  
    ) / 86400000.0  

    recency_score = math.exp(  
        -max(0, age_days) / 14.0  
    )  
else:  
    recency_score = 0.0  

w = DEFAULTS["weights"]  

return round(  
    w["time"] * time_score  
    + w["launches"] * launch_score  
    + w["recency"] * recency_score,  
    4  
)

def parse_ui_nodes(xml):
nodes = []

for match in re.finditer(  
    r'<node\b([^>]*)/>',  
    xml  
):  
    attrs = dict(  
        re.findall(  
            r'(\w+)="([^"]*)"',  
            match.group(1)  
        )  
    )  

    text = attrs.get("text", "")  
    desc = attrs.get(  
        "content-desc",  
        ""  
    )  
    bounds = attrs.get(  
        "bounds",  
        ""  
    )  

    bm = re.search(  
        r'(\d+),(\d+)(\d+),(\d+)',  
        bounds  
    )  

    if not bm:  
        continue  

    x1, y1, x2, y2 = map(  
        int,  
        bm.groups()  
    )  

    if not text and not desc:  
        continue  

    nodes.append({  
        "text": text,  
        "desc": desc,  
        "bounds": bounds,  
        "center": [  
            (x1 + x2) // 2,  
            (y1 + y2) // 2  
        ]  
    })  

return nodes

    start = time.monotonic()

    try:
        p = subprocess.Popen(

def cat(label,p):

def score(u,now):
    if not u:
        return 0.0

    tm=min(1,math.log1p(max(0,u[0])/60000)/math.log1p(600))
    la=min(1,math.log1p(max(0,u[2]))/math.log1p(100))
    recc=0

    if u[1] and u[1]<=now:
        age=(now-u[1])/86400000
        recc=math.exp(-math.log(2)*max(0,age)/CFG["half_life_days"])

    return round(.55*tm+.25*la+.20*recc,4)

def cc(uri,act,bind=(),where=None):
    a=["content",act,"--uri",q(uri)]
    for b in bind:
        a+=["--bind",q(b)]
    if where:
        a+=["--where",q(where)]
    return " ".join(a)

def slots(items,exclude):
    return list(dict.fromkeys(
        (
            int(x["screen"]),
            int(x["cellX"]),
            int(x["cellY"])

        for x in items
        if x["itemType"]==0
        and x["container"]==-100

    ))

def balanced(ss,n):
    if n==0:
        return []
    if len(ss)<n:
        return []

    cx=sum(s[1] for s in ss)/len(ss)
    cy=sum(s[2] for s in ss)/len(ss)
    rem=ss[:]

    a=min(
        rem,
        key=lambda s:(
            abs(s[1]-cx)+abs(s[2]-cy),
            s[0],
            s[1],
            s[2]
        )
    )
    out=[a]
    rem.remove(a)

    def dist(x,y):
        return (
            abs(x[0]-y[0])*4+
            abs(x[1]-y[1])+
            abs(x[2]-y[2])
        )

    while rem and len(out)<n:
        z=max(
            rem,
            key=lambda s:min(dist(s,a) for a in out)
        )
        out.append(z)
        rem.remove(z)

    return out

def verify(pp,provider):
    ok,o,e=run(f"content query --uri {q(provider)}",30)
    if not ok or "Row:" not in o:
        raise RuntimeError("Post-apply verification could not read launcher database.")

    by={}
    for x in rows(o):
        try:
            by[int(num(x.get("_id"),-1))]=x
        except:
            pass

    for f in pp["folders"]:
        x=by.get(f["id"])
        if not x:
            raise RuntimeError(f"Verification failed for folder {f['name']}.")

        if int(num(x.get("itemType"),-1))!=2:
            raise RuntimeError(f"Verification failed for folder {f['name']}.")

        if int(num(x.get("container"),-999))!=-100:
            raise RuntimeError(f"Verification failed for folder {f['name']}.")

        pos=(
            int(num(x.get("screen"),-1)),
            int(num(x.get("cellX"),-1)),
            int(num(x.get("cellY"),-1))
        )

        if pos!=(f["screen"],f["cellX"],f["cellY"]):
            raise RuntimeError(f"Verification failed for folder position {f['name']}.")

        for a in f["apps"]:
            x=by.get(a["id"])
            if(
                not x
                or int(num(x.get("container"),-999))!=f["id"]
                or int(num(x.get("screen"),-1))!=0
                or int(num(x.get("cellX"),-1))!=a["cellX"]
                or int(num(x.get("cellY"),-1))!=a["cellY"]
            ):
                raise RuntimeError(f"Verification failed for app {a['title']}.")

    for a in pp["outside"]:
        x=by.get(a["id"])

        if a["nested"]:
            if not x or int(num(x.get("container"),-999))!=-100:
                raise RuntimeError(f"Verification failed for outside app {a['title']}.")

        if(
            not x
            or int(num(x.get("container"),-999))!=-100
            or (
                int(num(x.get("screen"),-1)),
                int(num(x.get("cellX"),-1)),
                int(num(x.get("cellY"),-1))
            )!=(a["screen"],a["cellX"],a["cellY"])
        ):
            raise RuntimeError(f"Verification failed for outside app {a['title']}.")

def rollback(st):
    restart(st["launcher"])

    for x in st.get("items",[]):
        try:
            upd(
                st["provider"]+"?notify=true",
                x["id"],
                x["c"],
                x["s"],
                x["x"],
                x["y"]
            )
        except:
            pass

    for x in st.get("new",[]):
        try:
            dele(st["provider"]+"?notify=true",x)
        except:
            pass

    for f in st.get("folders",[]):
        try:
            fupd(
                st["provider"]+"?notify=true",
                f["id"],
                f["title"],
                f["s"],
                f["x"],
                f["y"]
            )
        except:
            pass

    restart(st["launcher"])

def apply(pp,st):
    ok,o,e=run(
        cc(
            st["provider"],
            "update",
            ["cellX:i:0"],
            "_id=0"
        ),
        30
    )

    if not ok:
        raise RuntimeError(
            "Launcher provider is readable but not writable through Shizuku: "
            +(e.strip() or o.strip())
        )

    restart(st["launcher"])

    try:
        exist={f["id"] for f in st["folders"]}

        for a in pp["outside"]:
            if a["nested"]:
                upd(
                    st["provider"]+"?notify=true",
                    a["id"],
                    -100,
                    a["screen"],
                    a["cellX"],
                    a["cellY"]
                )

        for f in pp["folders"]:
            if f["id"] in exist:
                fupd(
                    st["provider"]+"?notify=true",
                    f["id"],
                    f["name"],
                    f["screen"],
                    f["cellX"],
                    f["cellY"]
                )
            else:
                ins(
                    st["provider"]+"?notify=true",
                    f["id"],
                    f["name"],
                    f["screen"],
                    f["cellX"],
                    f["cellY"]
                )

            for a in f["apps"]:
                upd(
                    st["provider"]+"?notify=true",
                    a["id"],
                    f["id"],
                    0,
                    a["cellX"],
                    a["cellY"]
                )

        used={f["id"] for f in pp["folders"]}

        for f in st["folders"]:
            if f["id"] not in used:
                dele(
                    st["provider"]+"?notify=true",
                    f["id"]
                )

        verify(pp,st["provider"])
        restart(st["launcher"])

    except:
        rollback(st)
        raise

    fp=[]

    for c,v in sorted(

        if c=="Other" or len(v)<CFG["folder_min_apps"]:
            loose+=v
            continue

        for i in range(0,len(v),CFG["folder_max_apps"]):
            ch=v[i:i+CFG["folder_max_apps"]]

            if len(ch)>=CFG["folder_min_apps"]:
                fp.append({
                    "name":
                        c if i==0
                        else f"{c} {i//CFG['folder_max_apps']+1}",
                    "apps":ch
                })
            else:
                loose+=ch

    if len(fp)>CFG["max_folders"]:
        fp.sort(key=lambda z:-len(z["apps"]))
        loose+=sum(
            (z["apps"] for z in fp[CFG["max_folders"]:]),

    out+=loose
    outids={x["id"] for x in out}

    print("[5/7] Layout...")
